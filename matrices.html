let chart = null; function calcularRegresion() { const x = document.getElementById("xdata").value.split(",").map(Number); const y = document.getElementById("ydata").value.split(",").map(Number); const n = x.length; let sumX = x.reduce((a,b) => a + b, 0); let sumY = y.reduce((a,b) => a + b, 0); let sumXY = x.reduce((a,b,i) => a + b * y[i], 0); let sumX2 = x.reduce((a,b) => a + b*b, 0); const m = (n * sumXY - sumX * sumY) / (n * sumX2 - sumX * sumX); const b = (sumY - m * sumX) / n; document.getElementById("resultadoRegresion").innerText = `y = ${m.toFixed(4)}x + ${b.toFixed(4)}`; const puntos = x.map((valor,i)=>({ x: valor, y: y[i] })); const minX = Math.min(...x); const maxX = Math.max(...x); const linea = [ {x:minX, y:m*minX + b}, {x:maxX, y:m*maxX + b} ]; if(chart){ chart.destroy(); } const ctx = document.getElementById("grafica"); chart = new Chart(ctx,{ type:"scatter", data:{ datasets:[ { label:"Datos", data:puntos }, { label:"Regresión", data:linea, type:"line", fill:false } ] }, options:{ scales:{ x:{ type:"linear" } } } }); }
